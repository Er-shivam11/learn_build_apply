# Module 1: Traversal

Traversal means visiting every element exactly once to perform some operation.

```python
nums = [4, 9, 2, 15, 7, 11]

# Maximum
current = nums[0]
for i in nums:
    if i > current:
        current = i
print("Max:", current)

# Minimum
current = nums[0]
for i in nums:
    if i < current:
        current = i
print("Min:", current)

# Sum
current = 0
for i in nums:
    current += i
print("Sum:", current)

# Count Even
count = 0
for i in nums:
    if i % 2 == 0:
        count += 1
print("Even Count:", count)

# Count Odd
count = 0
for i in nums:
    if i % 2 == 1:
        count += 1
print("Odd Count:", count)

# Count Positive and Negative
nums = [4, 9, -2, 15, 7, 11]

positive = 0
negative = 0

for i in nums:
    if i > 0:
        positive += 1
    else:
        negative += 1

print("Positive:", positive)
print("Negative:", negative)
```

# Module 2: Searching

Searching problems involve finding values, positions, frequencies, or patterns inside collections.

```python
nums = [4, 9, 15, 2, 15, 7, 11, 15]
target = 15

# Find Index
def find_index(nums, target):
    for idx, i in enumerate(nums):
        if i == target:
            return idx
    return -1

print("Find Index:", find_index(nums, target))


# First Occurrence
def first_occ(nums, target):
    for idx, i in enumerate(nums):
        if i == target:
            return idx
    return -1

print("First Occurrence:", first_occ(nums, target))


# Last Occurrence
def last_occ(nums, target):
    answer = -1
    for idx, i in enumerate(nums):
        if i == target:
            answer = idx
    return answer

print("Last Occurrence:", last_occ(nums, target))


# Count Occurrence
def count_occ(nums, target):
    count = 0
    for i in nums:
        if i == target:
            count += 1
    return count

print("Count Occurrence:", count_occ(nums, target))


# All Occurrences
def all_occ(nums, target):
    result = []

    for idx, i in enumerate(nums):
        if i == target:
            result.append(idx)

    return result

print("All Occurrences:", all_occ(nums, target))


# Contains Duplicate
nums = [4, 9, 4, 2, 15, 7, 11, 9]

seen = []

for num in nums:
    if num in seen:
        print("Duplicate Found:", True)
        break
    else:
        seen.append(num)


# Missing Number
nums = [1, 2, 3, 5]

for i in range(1, len(nums) + 2):
    if i not in nums:
        print("Missing Number:", i)
```

# Module 3: Modification

Modification problems change the original array or create a transformed version.

```python
nums = [4, 9, 2, 15, 7, 11]

# Multiply by 2
result = []

for i in nums:
    result.append(i * 2)

print("Multiply by 2:", result)


# Replace Negative with 0
nums = [-4, 9, 2, 15, -7, 11]

for idx, i in enumerate(nums):
    if i < 0:
        nums[idx] = 0

print("Replace Negative:", nums)


# Square Every Element
nums = [-4, 9, 2, 15, -7, 11]

for idx, i in enumerate(nums):
    nums[idx] = i * i

print("Square Elements:", nums)


# Remove Negatives
nums = [-4, 9, 2, 15, -7, 11]

new = []

for i in nums:
    if i > 0:
        new.append(i)

print("Remove Negatives:", new)


# Remove Zeros
nums = [4, 0, 9, 0, 2, 0, 15, 7]

new = []

for i in nums:
    if i != 0:
        new.append(i)

print("Remove Zeros:", new)


# Move Zeros to End
nums = [4, 0, 9, 0, 2, 0, 15, 7]

length = len(nums)
new = []

for i in nums:
    if i != 0:
        new.append(i)

zero_count = length - len(new)

for _ in range(zero_count):
    new.append(0)

print("Move Zeros to End:", new)
```

# Module 4: Reversal & Swapping

These are classic Two Pointer problems and frequently appear in coding interviews.

```python
# Reverse Array
nums = [4, 9, 2, 15, 7, 11]

l = 0
r = len(nums) - 1

while l < r:
    nums[l], nums[r] = nums[r], nums[l]
    l += 1
    r -= 1

print("Reverse Array:", nums)


# Swap First and Last
nums = [4, 9, 2, 15, 7, 11]

l = 0
r = len(nums) - 1

nums[l], nums[r] = nums[r], nums[l]

print("Swap First Last:", nums)


# Swap Adjacent Pairs
nums = [4, 9, 2, 15, 7, 11]

for i in range(0, len(nums) - 1, 2):
    nums[i], nums[i + 1] = nums[i + 1], nums[i]

print("Swap Adjacent:", nums)


# Palindrome Check
s = "madam"

l = 0
r = len(s) - 1

is_palin = True

while l < r:
    if s[l] != s[r]:
        is_palin = False
        break

    l += 1
    r -= 1

print("Palindrome:", is_palin)


# Reverse String
s = "shivam"

result = ""

for ch in s:
    result = ch + result

print("Reverse String:", result)


# Reverse Vowels
s = "hello"
vowels = "aeiouAEIOU"

chars = list(s)

l = 0
r = len(chars) - 1

while l < r:

    if chars[l] not in vowels:
        l += 1
        continue

    if chars[r] not in vowels:
        r -= 1
        continue

    chars[l], chars[r] = chars[r], chars[l]

    l += 1
    r -= 1

print("Reverse Vowels:", "".join(chars))


# Reverse Words
s = "I love Python"

words = s.split()

l = 0
r = len(words) - 1

while l < r:
    words[l], words[r] = words[r], words[l]
    l += 1
    r -= 1

result = " ".join(words)

print("Reverse Words:", result)
```

# ------------------------------------------------------------
# 1. Left Rotate by 1
# Input : [1,2,3,4,5]
# Output: [2,3,4,5,1]
# ------------------------------------------------------------

nums = [1, 2, 3, 4, 5]

first = nums[0]

for i in range(len(nums) - 1):
    nums[i] = nums[i + 1]

nums[-1] = first

print("Left Rotate by 1:", nums)


# ------------------------------------------------------------
# 2. Right Rotate by 1
# Input : [1,2,3,4,5]
# Output: [5,1,2,3,4]
# ------------------------------------------------------------

nums = [1, 2, 3, 4, 5]

last = nums[-1]

for i in range(len(nums) - 1, 0, -1):
    nums[i] = nums[i - 1]

nums[0] = last

print("Right Rotate by 1:", nums)


# ------------------------------------------------------------
# 3. Left Rotate by K
# Input : [1,2,3,4,5], k=2
# Output: [3,4,5,1,2]

nums = [1, 2, 3, 4, 5]
k = 2

result = []

for i in range(k, len(nums)):
    result.append(nums[i])

for i in range(k):
    result.append(nums[i])

print("Left Rotate by K:", result)

# ------------------------------------------------------------
# 3. Range Sum Query
# Input: nums=[2,4,6,8,10], left=1, right=3
# Output: 18
# ------------------------------------------------------------

nums = [2, 4, 6, 8, 10]
left = 1
right = 3

running = 0
prefix = []

for i in nums:
    running += i
    prefix.append(running)

if left == 0:
    answer = prefix[right]
else:
    answer = prefix[right] - prefix[left - 1]

print("Range Sum:", answer)


# ------------------------------------------------------------
# 4. Pivot Index
# Input: [1,7,3,6,5,6]
# Output: 3
# ------------------------------------------------------------

nums = [1, 7, 3, 6, 5, 6]

total = sum(nums)
left_sum = 0

for i in range(len(nums)):
    right_sum = total - left_sum - nums[i]

    if left_sum == right_sum:
        print("Pivot Index:", i)
        break

    left_sum += nums[i]


# ------------------------------------------------------------
# 5. Equilibrium Index
# Input: [-7,1,5,2,-4,3,0]
# Output: 3
# ------------------------------------------------------------

nums = [-7, 1, 5, 2, -4, 3, 0]

total = sum(nums)
left_sum = 0

for i in range(len(nums)):
    right_sum = total - left_sum - nums[i]

    if left_sum == right_sum:
        print("Equilibrium Index:", i)
        break

    left_sum += nums[i]

# 6. Product Except Self
# Input: [1,2,3,4]
# Output: [24,12,8,6]
# ------------------------------------------------------------

nums = [1, 2, 3, 4]
result = []

for i in range(len(nums)):
    product = 1

    for j in range(len(nums)):
        if i != j:
            product *= nums[j]

    result.append(product)

print("Product Except Self:", result)

# 7. Count Subarrays With Sum K

# ------------------------------------------------------------

nums = [1, 1, 1]
k = 2
count = 0

for i in range(len(nums)):
    current_sum = 0

    for j in range(i, len(nums)):

        if current_sum == k:
